cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rm_aa_monitor_client LANGUAGES CXX)
endif()
find_package(Threads REQUIRED)
add_library(monitor_client STATIC monitor.cpp)
target_compile_features(monitor_client PUBLIC cxx_std_17)
target_compile_definitions(monitor_client PUBLIC RM_AA_MONITOR)
target_include_directories(monitor_client PUBLIC include PRIVATE thirdparty)
target_link_libraries(monitor_client PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(monitor_client PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
  add_executable(monitor_probe ../tests/monitor_probe.cpp)
  target_link_libraries(monitor_probe PRIVATE monitor_client)
endif()
