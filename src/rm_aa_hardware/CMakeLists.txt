cmake_minimum_required(VERSION 3.16)
project(rm_aa_hardware LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(RM_AA_HARDWARE_STANDALONE "Build protocol tests without ROS or serial hardware" OFF)
# One source of truth: use the exact protocol and transport supplied to training.
set(RMV_SERIAL "${CMAKE_CURRENT_SOURCE_DIR}/../../training/serialPort")
if(NOT EXISTS "${RMV_SERIAL}/include/Communication.hpp")
  message(FATAL_ERROR "Build from the complete RM_AA checkout: training/serialPort is required")
endif()
add_library(hardware_protocol src/protocol.cpp)
target_include_directories(hardware_protocol PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${RMV_SERIAL}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(hardware_protocol PRIVATE /W4 /utf-8)
else()
  target_compile_options(hardware_protocol PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(RM_AA_HARDWARE_STANDALONE)
  enable_testing()
  find_package(GTest REQUIRED)
  add_executable(test_protocol test/test_protocol.cpp)
  target_link_libraries(test_protocol hardware_protocol GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(test_protocol)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The serial hardware node requires Linux; use the standalone option for protocol tests")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Threads REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/GimbalFeedback.msg" "msg/TargetCommand.msg" DEPENDENCIES std_msgs)
add_library(hardware_serial STATIC ${RMV_SERIAL}/src/SerialPort.cpp)
target_include_directories(hardware_serial PUBLIC ${RMV_SERIAL}/include)
add_executable(hardware_node src/hardware_node.cpp)
ament_target_dependencies(hardware_node rclcpp rcl_interfaces std_msgs sensor_msgs geometry_msgs trajectory_msgs tf2_ros)
rosidl_get_typesupport_target(hardware_typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(hardware_node hardware_protocol hardware_serial Threads::Threads "${hardware_typesupport}")
target_compile_options(hardware_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS hardware_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_protocol test/test_protocol.cpp)
  target_link_libraries(test_protocol hardware_protocol)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME hardware_pty COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_hardware_pty.py $<TARGET_FILE:hardware_node>)
  set_tests_properties(hardware_pty PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
