cmake_minimum_required(VERSION 3.16)
project(rm_aa_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RM_AA_CONTROL_STANDALONE "Build only the ROS-independent algorithms and tests" OFF)
if(NOT RM_AA_CONTROL_STANDALONE)
  find_package(ament_cmake REQUIRED)
  find_package(eigen3_cmake_module REQUIRED)
endif()
find_package(Eigen3 REQUIRED)
add_library(camera_control_core src/tracker.cpp src/mpc.cpp)
target_include_directories(camera_control_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(camera_control_core PUBLIC Eigen3::Eigen)
if(MSVC)
  target_compile_options(camera_control_core PRIVATE /W4)
else()
  target_compile_options(camera_control_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(RM_AA_CONTROL_STANDALONE)
  enable_testing()
  find_package(GTest REQUIRED)
  add_executable(test_control test/test_control.cpp)
  target_link_libraries(test_control camera_control_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(test_control)
  return()
endif()
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rm_aa_hardware REQUIRED)
add_executable(control_node src/control_node.cpp)
target_link_libraries(control_node camera_control_core)
ament_target_dependencies(control_node rclcpp rcl_interfaces geometry_msgs sensor_msgs
  nav_msgs trajectory_msgs std_msgs tf2_ros rm_aa_hardware)
install(TARGETS camera_control_core EXPORT camera_control_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS control_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_control test/test_control.cpp)
  target_link_libraries(test_control camera_control_core)
endif()
ament_export_targets(camera_control_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 eigen3_cmake_module)
ament_package()
