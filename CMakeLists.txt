cmake_minimum_required(VERSION 3.16)
project(RM_AA_Training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(ENABLE_MONITOR_CLIENT "Send training metrics to the FastAPI monitor" ON)
option(BUILD_TESTING "Build algorithm and protocol tests" ON)
find_package(Eigen3 REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio calib3d dnn video)
find_package(Threads REQUIRED)
set(VISION_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/src/rm_aa_vision)
add_library(perception_core STATIC
  ${VISION_ROOT}/src/source.cpp ${VISION_ROOT}/src/preprocessor.cpp
  ${VISION_ROOT}/src/detector.cpp ${VISION_ROOT}/src/classifier.cpp
  ${VISION_ROOT}/src/pose_solver.cpp ${VISION_ROOT}/src/image_tracker.cpp)
target_include_directories(perception_core PUBLIC ${VISION_ROOT}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(perception_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
include(${VISION_ROOT}/cmake/Hikrobot.cmake)
rm_aa_enable_hikrobot(perception_core)
add_library(camera_control_core STATIC src/rm_aa_control/src/tracker.cpp src/rm_aa_control/src/mpc.cpp)
target_include_directories(camera_control_core PUBLIC src/rm_aa_control/include)
target_link_libraries(camera_control_core PUBLIC Eigen3::Eigen)
add_library(training_core STATIC training/src/AppSupport.cpp training/src/AutoAimTypes.cpp training/src/StudentTasks.cpp)
target_include_directories(training_core PUBLIC training/include training/serialPort/include src/rm_aa_webUI/client/include)
target_link_libraries(training_core PUBLIC perception_core camera_control_core)
if(ENABLE_MONITOR_CLIENT)
  add_subdirectory(src/rm_aa_webUI/client)
  target_link_libraries(training_core PUBLIC monitor_client)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_subdirectory(training/serialPort)
  add_executable(rm_aa_training training/main.cpp)
  target_link_libraries(rm_aa_training PRIVATE training_core serialPort Threads::Threads)
else()
  message(STATUS "Linux required for the serial/pthread executable; algorithm tests remain available")
endif()
if(BUILD_TESTING)
  enable_testing()
  find_package(GTest REQUIRED)
  add_executable(test_control src/rm_aa_control/test/test_control.cpp)
  target_link_libraries(test_control PRIVATE camera_control_core GTest::gtest_main)
  if(EXISTS "${VISION_ROOT}/test/test_perception.cpp")
    add_executable(test_perception ${VISION_ROOT}/test/test_perception.cpp)
    target_link_libraries(test_perception PRIVATE perception_core GTest::gtest_main)
    target_compile_definitions(test_perception PRIVATE MODEL_DIRECTORY="${VISION_ROOT}/models")
  endif()
  add_executable(test_scaffold training/tests/test_scaffold.cpp)
  target_link_libraries(test_scaffold PRIVATE training_core)
  if(MSVC)
    target_compile_options(test_scaffold PRIVATE /UNDEBUG)
  else()
    target_compile_options(test_scaffold PRIVATE -UNDEBUG)
  endif()
  add_test(NAME scaffold_safety COMMAND test_scaffold)
  include(GoogleTest)
  gtest_discover_tests(test_control)
  if(TARGET test_perception)
    gtest_discover_tests(test_perception)
  endif()
  add_executable(test_training training/tests/test_training.cpp)
  target_link_libraries(test_training PRIVATE training_core GTest::gtest_main)
  target_compile_definitions(test_training PRIVATE MODEL_DIRECTORY="${VISION_ROOT}/models")
  gtest_discover_tests(test_training)
  if(TARGET rm_aa_training)
    add_test(NAME cli_help COMMAND rm_aa_training --help)
  endif()
endif()
