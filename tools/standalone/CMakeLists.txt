cmake_minimum_required(VERSION 3.16)
project(rm_aa_core_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CORE_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../src/rm_aa_vision)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio calib3d dnn video)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
find_package(GTest REQUIRED)
add_library(perception_core
  ${CORE_ROOT}/src/preprocessor.cpp ${CORE_ROOT}/src/detector.cpp
  ${CORE_ROOT}/src/classifier.cpp ${CORE_ROOT}/src/pose_solver.cpp
  ${CORE_ROOT}/src/image_tracker.cpp ${CORE_ROOT}/src/source.cpp)
target_include_directories(perception_core PUBLIC ${CORE_ROOT}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(perception_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
include(${CORE_ROOT}/cmake/Hikrobot.cmake)
rm_aa_enable_hikrobot(perception_core)
target_compile_options(perception_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(perception_cli ${CORE_ROOT}/src/perception_cli.cpp)
target_link_libraries(perception_cli perception_core nlohmann_json::nlohmann_json)
if(EXISTS "${CORE_ROOT}/test/test_perception.cpp")
add_executable(test_perception ${CORE_ROOT}/test/test_perception.cpp)
target_link_libraries(test_perception perception_core GTest::gtest_main)
target_compile_definitions(test_perception PRIVATE MODEL_DIRECTORY="${CORE_ROOT}/models")
enable_testing()
include(GoogleTest)
gtest_discover_tests(test_perception)
endif()
